import React, { useEffect } from 'react';
import { Stack } from 'expo-router';
import { StatusBar } from 'expo-status-bar';
import { View, Text, StyleSheet } from 'react-native';
import {
  useFonts,
  SpaceMono_400Regular,
  SpaceMono_700Bold,
} from '@expo-google-fonts/space-mono';
import * as SplashScreen from 'expo-splash-screen';
import { SafeAreaProvider } from 'react-native-safe-area-context';
import { GameProvider } from '../context/GameContext';
import { Colors } from '../constants/colors';
import Ogretici from '../components/Ogretici';
import { sesBaslat } from '../services/soundManager';
import { reklamlariBaslat } from '../services/adManager';

SplashScreen.preventAutoHideAsync();

class ErrorBoundary extends React.Component<
  { children: React.ReactNode },
  { hasError: boolean; error: string }
> {
  constructor(props: { children: React.ReactNode }) {
    super(props);
    this.state = { hasError: false, error: '' };
  }

  static getDerivedStateFromError(error: Error) {
    return {
      hasError: true,
      error: error?.message ?? 'Bilinmeyen hata',
    };
  }

  render() {
    if (this.state.hasError) {
      return (
        <View style={ebStyles.container}>
          <Text style={ebStyles.title}>Hata Oluştu</Text>
          <Text style={ebStyles.message}>{this.state.error}</Text>
        </View>
      );
    }

    return this.props.children;
  }
}

const ebStyles = StyleSheet.create({
  container: {
    flex: 1,
    justifyContent: 'center',
    alignItems: 'center',
    backgroundColor: Colors.background,
    padding: 24,
  },
  title: {
    fontSize: 22,
    color: Colors.danger,
    fontWeight: 'bold',
    marginBottom: 12,
  },
  message: {
    fontSize: 14,
    color: Colors.textPrimary,
    textAlign: 'center',
  },
});

export default function RootLayout() {
  const [fontsLoaded, fontError] = useFonts({
    SpaceMono: SpaceMono_400Regular,
    SpaceMonoBold: SpaceMono_700Bold,
  });

  useEffect(() => {
    const timer = setTimeout(() => {
      SplashScreen.hideAsync();
    }, 3000);

    if (fontsLoaded || fontError) {
      SplashScreen.hideAsync();
      clearTimeout(timer);
    }

    return () => clearTimeout(timer);
  }, [fontsLoaded, fontError]);

  // Ses ve reklam sistemini bir kez başlat
  useEffect(() => {
    sesBaslat();
    reklamlariBaslat();
  }, []);

  if (!fontsLoaded && !fontError) {
    return null;
  }

  return (
    <ErrorBoundary>
      <SafeAreaProvider>
        <GameProvider>
          <StatusBar style="light" />

          <Stack
            screenOptions={{
              headerShown: false,
              contentStyle: {
                backgroundColor: Colors.background,
              },
            }}
          >
            <Stack.Screen name="index" />
            <Stack.Screen name="intro" />
            <Stack.Screen name="tabs" />
            <Stack.Screen name="gameover" />
            <Stack.Screen name="+not-found" />
          </Stack>

          <Ogretici />
        </GameProvider>
      </SafeAreaProvider>
    </ErrorBoundary>
  );
}

// BEŞLİ PATRON — components/OyunHUD.tsx

// Gün / Enerji / Nakit / Prestij — tüm sekmelerde sabit üst bar
// ============================================
import React from 'react';
import { View, Text, StyleSheet, useWindowDimensions } from 'react-native';
import { usePathname } from 'expo-router';
import { useSafeAreaInsets } from 'react-native-safe-area-context';
import { useGame } from '../context/GameContext';
import { MAX_ENERJI } from '../constants/gameConfig';

function gs(v: unknown, yedek = 0): number {
  const n = Number(v);
  return Number.isFinite(n) ? n : yedek;
}

function paraFormat(n: number): string {
  return Math.round(n).toLocaleString('tr-TR');
}

export default function OyunHUD() {
  const { state } = useGame();
  const pathname = usePathname() ?? '';
  const { width } = useWindowDimensions();
  const insets = useSafeAreaInsets();
  const k = Math.max(0.85, Math.min(1.4, width / 375));

  // Yalnızca sekme ekranlarında görünür (index/intro/gameover'da gizlenir)
  if (!state || !pathname.startsWith('/tabs')) return null;

  const maxEnerji = gs(MAX_ENERJI, 100);
  const enerji = Math.max(0, Math.min(maxEnerji, gs(state.enerji)));
  const oran = maxEnerji > 0 ? enerji / maxEnerji : 0;
  const nakit = gs(state.nakit);
  const prestij = gs(state.prestij);
  const gun = gs(state.gun, 1);

  return (
    <View
      style={[
        st.dis,
        {
          paddingHorizontal: 12 * k,
          paddingTop: Math.max(6 * k, insets.top + 2),
          paddingBottom: 6 * k,
        },
      ]}
    >
      <View
        style={[
          st.kapsul,
          { borderRadius: 14 * k, paddingVertical: 7 * k, paddingHorizontal: 10 * k },
        ]}
      >
        <View style={st.hucre}>
          <Text style={[st.ikon, { fontSize: 15 * k }]}>📅</Text>
          <Text style={[st.deger, { fontSize: 13 * k }]}>Gün {gun}</Text>
        </View>

        <View style={[st.hucre, { flex: 1 }]}>
          <Text style={[st.ikon, { fontSize: 15 * k }]}>⚡</Text>
          <View style={st.enerjiKutu}>
            <Text style={[st.deger, { fontSize: 13 * k }]}>{enerji}/{maxEnerji}</Text>
            <View style={[st.enerjiIzgi, { height: 5 * k, borderRadius: 3 * k }]}>
              <View
                style={{
                  height: '100%',
                  width: `${Math.round(oran * 100)}%`,
                  borderRadius: 3 * k,
                  backgroundColor: '#4ADE80',
                }}
              />
            </View>
          </View>
        </View>

        <View style={st.hucre}>
          <Text style={[st.ikon, { fontSize: 15 * k }]}>💰</Text>
          <Text style={[st.deger, { fontSize: 13 * k }]}>{paraFormat(nakit)}₺</Text>
        </View>

        <View style={st.hucre}>
          <Text style={[st.ikon, { fontSize: 15 * k }]}>⭐</Text>
          <Text style={[st.deger, { fontSize: 13 * k }]}>{prestij}</Text>
        </View>
      </View>
    </View>
  );
}

const st = StyleSheet.create({
  dis: { backgroundColor: 'rgba(10,12,24,0.96)' },
  kapsul: {
    flexDirection: 'row',
    alignItems: 'center',
    backgroundColor: 'rgba(30,41,59,0.85)',
    borderWidth: 1,
    borderColor: 'rgba(255,255,255,0.10)',
  },
  hucre: { flexDirection: 'row', alignItems: 'center', paddingHorizontal: 8 },
  ikon: { marginRight: 4 },
  deger: { color: '#FFFFFF', fontWeight: '700', fontFamily: 'SpaceMono' },
  enerjiKutu: { marginLeft: 2, width: 68 },
  enerjiIzgi: { backgroundColor: 'rgba(255,255,255,0.15)', marginTop: 3, overflow: 'hidden' },

// components/OlayModal.tsx — Sardur Technologies
// Gün başlarken olay varsa tam ekran kare olarak açılır; seçim yapmadan gün ilerlemez.
import React from 'react';
import { Modal, View, Text, TouchableOpacity, StyleSheet, useWindowDimensions } from 'react-native';
import { usePathname } from 'expo-router';
import { useGame } from '../context/GameContext';
import { OLAY_HAVUZU } from '../constants/gameConfig';

function gs(v: unknown, yedek = 0): number {
  const n = Number(v);
  return Number.isFinite(n) ? n : yedek;
}

export default function OlayModal() {
  const { state, cozOlay } = useGame();
  const pathname = usePathname() ?? '';
  const { width } = useWindowDimensions();
  const k = Math.max(0.85, Math.min(1.4, width / 375));

  const olayId = state?.bugunOlayId;
  const sablon: any = OLAY_HAVUZU?.find((o: any) => o?.id === olayId) as any;

  if (!state || state.isGameOver || !olayId || state.olayCozuldu || !sablon) return null;
  if (!pathname.startsWith('/tabs')) return null;

  const baslik = sablon.ad ?? sablon.baslik ?? sablon.title ?? 'Günün Olayı';
  const aciklama = sablon.aciklama ?? sablon.description ?? sablon.metin ?? '';
  const secenekler: any[] = Array.isArray(sablon.secenekler) ? sablon.secenekler : [];
  if (secenekler.length === 0) return null; // seçeneksiz olay modalda zorlamaz

  const etkiOzeti = (s: any): string => {
    const e = s?.etki ?? {};
    const parcalar: string[] = [];
    const nakit = gs(e.nakit);
    if (nakit !== 0) parcalar.push(`${nakit > 0 ? '+' : ''}${nakit.toLocaleString('tr-TR')}₺`);
    const enerji = gs(e.enerji);
    if (enerji !== 0) parcalar.push(`${enerji > 0 ? '+' : ''}${enerji} enerji`);
    const jeton = gs(e.jeton);
    if (jeton !== 0) parcalar.push(`${jeton > 0 ? '+' : ''}${jeton} jeton`);
    const prestij = gs(e.prestij);
    if (prestij !== 0) parcalar.push(`${prestij > 0 ? '+' : ''}${prestij} prestij`);
    const mem = gs(e.memnuniyet);
    if (mem !== 0) parcalar.push(`${mem > 0 ? '+' : ''}${mem} memnuniyet`);
    return parcalar.join(' · ');
  };

  return (
    <Modal transparent animationType="fade" visible statusBarTranslucent>
      <View style={st.arka}>
        <View style={[st.kart, { width: Math.min(340 * k, width - 32), borderRadius: 18 * k, padding: 18 * k }]}>
          <Text style={[st.baslik, { fontSize: 19 * k }]}>🎲 {baslik}</Text>
          {aciklama ? <Text style={[st.aciklama, { fontSize: 14 * k }]}>{aciklama}</Text> : null}
          {secenekler.map((s: any, i: number) => {
            const etiket = s?.ad ?? s?.metin ?? s?.etiket ?? s?.label ?? `Seçenek ${i + 1}`;
            const ozet = etkiOzeti(s);
            return (
              <TouchableOpacity
                key={i}
                style={[st.secenek, { paddingVertical: 12 * k, borderRadius: 12 * k, paddingHorizontal: 14 * k }]}
                onPress={() => cozOlay(i)}
              >
                <Text style={[st.secenekYazi, { fontSize: 14 * k }]}>{etiket}</Text>
                {ozet ? <Text style={[st.secenekOzet, { fontSize: 12 * k }]}>{ozet}</Text> : null}
              </TouchableOpacity>
            );
          })}
        </View>
      </View>
    </Modal>
  );
}

const st = StyleSheet.create({
  arka: {
    flex: 1,
    backgroundColor: 'rgba(0,0,0,0.75)',
    justifyContent: 'center',
    alignItems: 'center',
  },
  kart: {
    backgroundColor: '#111827',
    borderWidth: 1,
    borderColor: 'rgba(245,166,35,0.5)',
  },
  baslik: { color: '#F5A623', fontWeight: 'bold', textAlign: 'center', marginBottom: 10 },
  aciklama: { color: '#E5E7EB', lineHeight: 20, marginBottom: 12 },
  secenek: {
    backgroundColor: '#1E293B',
    borderWidth: 1,
    borderColor: 'rgba(255,255,255,0.08)',
    marginTop: 10,
  },
  secenekYazi: { color: '#FFFFFF', fontWeight: '700' },
  secenekOzet: { color: '#9CA3AF', marginTop: 3 },
});
