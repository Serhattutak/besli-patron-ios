import React, { useState } from 'react';
import {
  View,
  Text,
  Pressable,
  ScrollView,
  StyleSheet,
  Platform,
  Modal,
} from 'react-native';
import { router } from 'expo-router';
import { useGame } from '../../context/GameContext';
import { Colors } from '../../constants/colors';
import {
  JETON_ENERJI_COST,
  JETON_ENERJI_GAIN,
  MAX_ENERJI,
  CITIES,
} from '../../constants/gameConfig';

export default function ProfilScreen() {
  const {
    state,
    useJeton,
    manualSave,
    resetGame,
    unlockCity,
    switchCity,
    devHediye,
  } = useGame();

  const [message, setMessage] = useState('');
  const [showResetConfirm, setShowResetConfirm] = useState(false);
  const [confirmCity, setConfirmCity] = useState<string | null>(null);
  const [showDevPanel, setShowDevPanel] = useState(false);
  const [futbolcuTik, setFutbolcuTik] = useState(0);

  const font = Platform.select({
    ios: 'SpaceMono',
    android: 'SpaceMono',
    web: 'SpaceMono, monospace',
  });

  const jeton = state?.jeton ?? 0;
  const enerji = state?.enerji ?? 0;
  const prestij = state?.prestij ?? 0;
  const nakit = state?.nakit ?? 0;

  const canUseJeton =
    jeton >= (JETON_ENERJI_COST ?? 20) &&
    enerji < (MAX_ENERJI ?? 100);

  const showMsg = (msg: string) => {
    setMessage(msg);
    setTimeout(() => setMessage(''), 2500);
  };

  const handleFutbolcuTik = () => {
    const yeniTik = futbolcuTik + 1;

    if (yeniTik >= 5) {
      setShowDevPanel(true);
      setFutbolcuTik(0);
      showMsg('Geliştirici paneli açıldı! 🧪');
    } else {
      setFutbolcuTik(yeniTik);
    }
  };

  const handleUseJeton = () => {
    const err = useJeton();

    if (err) {
      showMsg(err);
    } else {
      showMsg(`+${JETON_ENERJI_GAIN ?? 30} Enerji yüklendi!`);
    }
  };

  const handleSave = async () => {
    await manualSave();
    showMsg('Oyun kaydedildi!');
  };

  const handleReset = async () => {
    setShowResetConfirm(false);
    await resetGame();
    router.replace('/intro');
  };

  const handleCityAction = (cityId: string) => {
    const cityData = state?.cities?.[cityId];

    if (cityData?.unlocked) {
      const err = switchCity(cityId);

      if (err) {
        showMsg(err);
      } else {
        showMsg('Şehir değiştirildi! 🏙️');
      }
    } else {
      setConfirmCity(cityId);
    }
  };

  const handleUnlockConfirm = () => {
    if (!confirmCity) return;

    const err = unlockCity(confirmCity);

    if (err) {
      showMsg(err);
    } else {
      showMsg('Şehir açıldı! 🎉');
    }

    setConfirmCity(null);
  };

  const handleGiftMoney = () => {
    devHediye(1000000, 0, 0, false);
    showMsg('+1.000.000₺ eklendi! 💰');
  };

  const handleGiftPrestige = () => {
    devHediye(0, 100, 0, false);
    showMsg('+100 prestij eklendi! ⭐');
  };

  const handleGiftJeton = () => {
    devHediye(0, 0, 50, false);
    showMsg('+50 jeton eklendi! 🔵');
  };

  const handleFillEnergy = () => {
    devHediye(0, 0, 0, true);
    showMsg('Enerji tamamen dolduruldu! ⚡');
  };

  const cityList = CITIES ?? [];
  const confirmCityConfig = confirmCity
    ? cityList.find((city: any) => city?.id === confirmCity)
    : null;

  return (
    <>
      <ScrollView
        style={styles.container}
        contentContainerStyle={styles.content}
      >
        {message ? (
          <View style={styles.messageBox}>
            <Text style={[styles.messageText, { fontFamily: font }]}>
              {message}
            </Text>
          </View>
        ) : null}

        <View style={styles.card}>
          <View style={styles.avatarRow}>
            <View style={styles.avatar}>
              <View style={styles.avatarHead} />
              <View style={styles.avatarBody} />
            </View>

            <View style={styles.charInfo}>
              <Pressable onPress={handleFutbolcuTik}>
                <Text style={[styles.charName, { fontFamily: font }]}>
                  Eski Futbolcu
                </Text>
              </Pressable>

              <Text style={[styles.charTitle, { fontFamily: font }]}>
                Halısaha Patronu
              </Text>
            </View>
          </View>
        </View>

        <View style={styles.card}>
          <Text style={[styles.cardTitle, { fontFamily: font }]}>
            📊 İstatistikler
          </Text>

          <View style={styles.statRow}>
            <Text style={[styles.statLabel, { fontFamily: font }]}>
              📅 Toplam Gün
            </Text>
            <Text style={[styles.statValue, { fontFamily: font }]}>
              {state?.gun ?? 1}
            </Text>
          </View>

          <View style={styles.statRow}>
            <Text style={[styles.statLabel, { fontFamily: font }]}>
              💰 Nakit
            </Text>
            <Text style={[styles.statValueGold, { fontFamily: font }]}>
              {nakit.toLocaleString('tr-TR')} ₺
            </Text>
          </View>

          <View style={styles.statRow}>
            <Text style={[styles.statLabel, { fontFamily: font }]}>
              💵 Toplam Kazanç
            </Text>
            <Text style={[styles.statValueGold, { fontFamily: font }]}>
              {(state?.toplamKazanc ?? 0).toLocaleString('tr-TR')} ₺
            </Text>
          </View>

          <View style={styles.statRow}>
            <Text style={[styles.statLabel, { fontFamily: font }]}>
              🏟️ Saha Kalitesi
            </Text>
            <Text style={[styles.statValue, { fontFamily: font }]}>
              {state?.sahaKalitesi ?? 4}/10
            </Text>
          </View>

          <View style={styles.statRow}>
            <Text style={[styles.statLabel, { fontFamily: font }]}>
              ⚡ Enerji
            </Text>
            <Text style={[styles.statValue, { fontFamily: font }]}>
              {enerji}/{MAX_ENERJI ?? 100}
            </Text>
          </View>

          <View style={styles.statRow}>
            <Text style={[styles.statLabel, { fontFamily: font }]}>
              ⭐ Prestij
            </Text>
            <Text style={[styles.statValuePrestige, { fontFamily: font }]}>
              {prestij}
            </Text>
          </View>

          <View style={styles.statRow}>
            <Text style={[styles.statLabel, { fontFamily: font }]}>
              🔵 Jeton
            </Text>
            <Text style={[styles.statValue, { fontFamily: font }]}>
              {jeton}
            </Text>
          </View>
        </View>

        <View style={styles.card}>
          <Text style={[styles.cardTitle, { fontFamily: font }]}>
            ⚡ Enerji ve Kayıt
          </Text>

          <Pressable
            style={[
              styles.actionButton,
              !canUseJeton && styles.disabledButton,
            ]}
            onPress={handleUseJeton}
            disabled={!canUseJeton}
          >
            <Text style={[styles.buttonText, { fontFamily: font }]}>
              🔵 Jeton Kullan — +{JETON_ENERJI_GAIN ?? 30} Enerji
            </Text>
          </Pressable>

          <Pressable style={styles.actionButton} onPress={handleSave}>
            <Text style={[styles.buttonText, { fontFamily: font }]}>
              💾 Oyunu Kaydet
            </Text>
          </Pressable>
        </View>

        <View style={styles.card}>
          <Text style={[styles.cardTitle, { fontFamily: font }]}>
            🏙️ Şehirler
          </Text>

          {cityList.map((city: any) => {
            const cityData = state?.cities?.[city?.id];
            const unlocked = cityData?.unlocked === true;
            const active = state?.activeSehir === city?.id;
            const unlockCost = Number(city?.unlockCost ?? 0);
            const requiredPrestij = Number(
              city?.requiredPrestij ?? city?.prestijCost ?? 0,
            );

            return (
              <View key={city?.id} style={styles.cityRow}>
                <View style={styles.cityInfo}>
                  <Text style={[styles.cityName, { fontFamily: font }]}>
                    {city?.emoji ?? '🏙️'} {city?.label ?? city?.id}
                  </Text>

                  {unlocked ? (
                    <Text style={[styles.cityStatus, { fontFamily: font }]}>
                      {active ? '✅ Aktif şehir' : '🔓 Açık'}
                    </Text>
                  ) : (
                    <Text style={[styles.cityLocked, { fontFamily: font }]}>
                      🔒 {unlockCost.toLocaleString('tr-TR')}₺ / ⭐{requiredPrestij}
                    </Text>
                  )}
                </View>

                <Pressable
                  style={[
                    styles.smallButton,
                    active && styles.disabledButton,
                  ]}
                  onPress={() => handleCityAction(city?.id)}
                  disabled={active}
                >
                  <Text style={[styles.smallButtonText, { fontFamily: font }]}>
                    {active ? 'Aktif' : unlocked ? 'Geç' : 'Aç'}
                  </Text>
                </Pressable>
              </View>
            );
          })}
        </View>

        <Pressable
          style={styles.resetButton}
          onPress={() => setShowResetConfirm(true)}
        >
          <Text style={[styles.resetButtonText, { fontFamily: font }]}>
            🗑️ Oyunu Sıfırla
          </Text>
        </Pressable>
      </ScrollView>

      <Modal
        visible={showDevPanel}
        transparent
        animationType="fade"
        onRequestClose={() => setShowDevPanel(false)}
      >
        <View style={styles.modalOverlay}>
          <View style={styles.devPanel}>
            <Text style={[styles.devTitle, { fontFamily: font }]}>
              🧪 Geliştirici Test Paneli
            </Text>

            <Text style={[styles.devDescription, { fontFamily: font }]}>
              Kilitli özellikleri test etmek için kullanılır.
            </Text>

            <Pressable style={styles.devButton} onPress={handleGiftMoney}>
              <Text style={[styles.devButtonText, { fontFamily: font }]}>
                💰 +1.000.000₺
              </Text>

function para(n: number): string {
  return `${Math.round(n).toLocaleString('tr-TR')}₺`;

function asamaEtiketi(asama: number): string {
  if (asama <= 0) return '🌱 Arsa Yok';
  if (asama === 1) return '🏠 Arsa Alındı';
  if (asama < 5) return '🚧 İnşaat';
  return '✅ Hizmette';
}

function asamaRengi(asama: number): string {
  if (asama <= 0) return RENK.kilit;
  if (asama === 1) return '#60A5FA';
  if (asama < 5) return RENK.vurgu;
  return RENK.basari;
}

// Sahadaki yükseltmelerin toplu etkisi — pitchConfig tanımlarından yerelde hesaplanır
function etkiHesapla(up: Record<string, number>) {
  let fiyat = 0, memnuniyet = 0, talep = 0, pasifGelir = 0, prestij = 0, enerjiIndirimi = 0, havaKoruma = false;
  Object.entries(up ?? {}).forEach(([id, seviye]) => {
    const d: PitchUpgradeDef | undefined = PITCH_UPGRADE_CONFIG[id];
    if (!d || !seviye) return;
    fiyat += (d.fiyatEki ?? 0) * seviye;
    memnuniyet += (d.memnuniyetEki ?? 0) * seviye;
    talep += (d.talepEki ?? 0) * seviye;
    pasifGelir += (d.gunlukGelirEki ?? 0) * seviye;
    prestij += (d.prestijEki ?? 0) * seviye;
    enerjiIndirimi += (d.enerjiIndirimi ?? 0) * seviye;
    if (d.havaKoruma) havaKoruma = true;
  });
  return { fiyat, memnuniyet, talep, pasifGelir, prestij, enerjiIndirimi, havaKoruma };
}

// Sponsor kartındaki şart satırları — sağlananlar ✅, eksikler ⬜
function sponsorKosulSatirlari(sp: SponsorAnlasma, anaSahaAsama: number, anaSahaUp: Record<string, number>): string[] {
  const satirlar: string[] = [];
  if (sp.kosul.asama != null) {
    satirlar.push(`${anaSahaAsama >= sp.kosul.asama ? '✅' : '⬜'} Ana saha aşama ${sp.kosul.asama}`);
  }
  if (sp.kosul.upgrade) {
    const d = PITCH_UPGRADE_CONFIG[sp.kosul.upgrade.id];
    const sev = anaSahaUp[sp.kosul.upgrade.id] ?? 0;
    satirlar.push(`${sev >= sp.kosul.upgrade.seviye ? '✅' : '⬜'} ${d?.ad ?? sp.kosul.upgrade.id} Sv.${sp.kosul.upgrade.seviye}`);
  }
  if (satirlar.length === 0) satirlar.push('✅ Şart yok');
  return satirlar;
}

        {/* === AKTİF SAHA YÖNETİM PANELİ === */}
        <View style={s.panel}>
          <View style={s.panelBaslikSatir}>
            <Text style={s.panelBaslik}>{tipBilgi.emoji} {aktif.ad} — Yönetim</Text>
            <View style={[s.durumRozet, { backgroundColor: asamaRengi(aktif.asama ?? 0) }]}>
              <Text style={s.durumRozetYazi}>{asamaEtiketi(aktif.asama ?? 0)}</Text>
            </View>
          </View>

          {/* === 2D BÜYÜK SAHA + CANLI MAÇ === */}
          <Saha2D
            asama={aktif.asama ?? 0}
            kalite={gosterilenKalite}
            sahaTipi={aktif.sahaTipi ?? 'yildiz'}
            canli={canliIzle && acik}
            upgrades={up}
          />
          <View style={s.canliSatir}>
            <TouchableOpacity style={s.canliButon} onPress={() => setCanliIzle(v => !v)} activeOpacity={0.8}>
              <Text style={s.canliButonYazi}>{canliIzle ? '⏸ Duraklat' : '▶️ Maçı İzle'}</Text>
            </TouchableOpacity>
            <Text style={s.canliBilgi}>
              {acik
                ? (canliIzle ? 'Oyuncular sahada — canlı maç!' : 'Maç duraklatıldı.')
                : 'Maç, hizmette olan sahalarda oynanır.'}
            </Text>
          </View>

          {/* === KURULU YAPILAR (gerçek yükseltme verileri) === */}
          {Object.entries(up).some(([, v]) => v > 0) ? (
            <View style={s.rozetSatir}>
              {Object.entries(up).filter(([, v]) => v > 0).map(([id, seviye]) => {
                const yd = PITCH_UPGRADE_CONFIG[id];
                if (!yd) return null;
                return (
                  <View key={id} style={s.rozet}>
                    <Text style={s.rozetYazi}>{yd.emoji} {yd.ad} · {seviye}/{yd.maksSeviye}</Text>
                  </View>
                );
              })}
            </View>
          ) : null}

          {/* İstatistikler */}
          <View style={s.istatistikSatiri}>
            <View style={s.istatistikKutu}>
              <Text style={s.istatistikDeger}>⭐ {gosterilenKalite}/10</Text>
              <Text style={s.istatistikEtiket}>Kalite</Text>
            </View>
            <View style={s.istatistikKutu}>
              <Text style={s.istatistikDeger}>{para(gunlukGider)}</Text>
              <Text style={s.istatistikEtiket}>Günlük Gider</Text>
            </View>
            <View style={s.istatistikKutu}>
              <Text style={s.istatistikDeger}>{acik ? para(pasifGelir) : '—'}</Text>
              <Text style={s.istatistikEtiket}>Pasif Gelir</Text>
            </View>
          </View>

          {/* İnşaat durumu */}
          {(aktif.asama ?? 0) >= 1 && (aktif.asama ?? 0) < 5 ? (
            <View style={s.kutu}>
              <Text style={s.kutuBaslik}>🚧 İnşaat — %{insaatYuzde} tamam</Text>
              <View style={s.bar}>
                <View style={[s.barDolgu, { width: `${Math.max(4, insaatYuzde)}%` }]} />
              </View>
              {sonrakiAsama ? (
                <View>
                  <Text style={s.kutuAltBaslik}>{sonrakiAsama.label}</Text>
                  <Text style={s.muted}>{sonrakiAsama.aciklama}</Text>
                  <TouchableOpacity style={s.butonYesil} onPress={insaatAdim} activeOpacity={0.8}>
                    <Text style={s.butonBeyaz}>
                      {state.nakit >= sonrakiAsama.cost
                        ? `İnşaatı İlerlet — ${para(sonrakiAsama.cost)} (⚡${sonrakiAsama.enerji})`
                        : `Nakit yetersiz (${para(sonrakiAsama.cost)})`}
                    </Text>
                  </TouchableOpacity>
                </View>
              ) : null}
            </View>
          ) : null}

          {/* Bakım (Yıldız) */}
          {yildiz ? (
            <View style={s.kutu}>
              <Text style={s.kutuBaslik}>🧹 Günlük Bakım</Text>
              <Text style={s.muted}>Bakım saha kalitesini korur; aksatırsan kalite düşer.</Text>
              <TouchableOpacity style={s.butonYesil} onPress={bakimYap} activeOpacity={0.8}>
                <Text style={s.butonBeyaz}>Bakım Yap — {para(BAKIM_UCRETI)} (⚡{ENERJI_BAKIM})</Text>
              </TouchableOpacity>
            </View>
          ) : null}

          {/* Yükseltme etkileri özeti */}
          <View style={s.etkiKutu}>
            <Text style={s.kutuBaslik}>📈 Yükseltme Etkileri</Text>
            <Text style={s.muted}>
              Fiyat +{Math.round(etki.fiyat * 100)}% · Memnuniyet +{etki.memnuniyet} · Prestij +{etki.prestij}
              {etki.pasifGelir > 0 ? ` · Pasif gelir ${para(etki.pasifGelir)}/gün` : ''}
              {etki.enerjiIndirimi > 0 ? ` · Enerji -${etki.enerjiIndirimi}` : ''}
              {etki.havaKoruma ? ' · ❄️ Hava koruması' : ''}
            </Text>
          </View>

          {/* 16 kalemlik saha yükseltme ızgarası */}
          <Text style={s.bolumBaslik}>Aktif Saha Yükseltmeleri</Text>
          <View style={s.yukseltmeIzgara}>
            {Object.values(PITCH_UPGRADE_CONFIG).map((ud: PitchUpgradeDef) => {
              const seviye = up[ud.id] ?? 0;
              const maliyet = pitchUpgradeMaliyeti(ud, seviye);
              return (
                <View key={ud.id} style={s.yukseltmeKart}>
                  <Text style={s.yukseltmeEmoji}>{ud.emoji}</Text>
                  <Text style={s.yukseltmeAd} numberOfLines={1}>{ud.ad}</Text>
                  <Text style={s.yukseltmeSeviye}>Seviye {seviye}/{ud.maksSeviye}</Text>
                  <Text style={s.yukseltmeEtki}>{ud.etki}</Text>
                  {maliyet != null ? (
                    <TouchableOpacity style={s.butonKucuk} onPress={() => yukseltAl(ud.id)} activeOpacity={0.8}>
                      <Text style={s.butonKucukYazi}>Yükselt — {para(maliyet)}</Text>
                    </TouchableOpacity>
                  ) : (
                    <View style={[s.butonKucuk, s.butonKucukPasif]}>
                      <Text style={s.butonKucukYazi}>Maks. seviye</Text>
                    </View>
                  )}
                </View>
              );
            })}
          </View>

          {/* Yıldız: tesis ana yükseltmeleri */}
          {yildiz ? (
            <View style={s.kutu}>
              <Text style={s.kutuBaslik}>🏗️ Tesis Yükseltmeleri</Text>
              {Object.entries(UPGRADE_CONFIG).map(([anahtar, cfgRaw]) => {
                const cfg = cfgRaw as any;
                const seviye = up[anahtar] ?? 0;
                const maks = Number(cfg?.maxLevel ?? 0);
                const maliyetListe = Array.isArray(cfg?.costs) ? cfg.costs : [];
                const maliyet = seviye < maks ? maliyetListe[seviye] : undefined;
                return (
                  <View key={anahtar} style={s.anaYukseltmeSatir}>
                    <View style={s.anaYukseltmeBilgi}>
                      <Text style={s.anaYukseltmeAd}>
                        {cfg?.emoji ?? '🔧'} {cfg?.ad ?? cfg?.baslik ?? anahtar}{' '}
                        <Text style={s.muted}>(Seviye {seviye}/{maks || '?'})</Text>
                      </Text>
                      {cfg?.aciklama ? <Text style={s.muted}>{String(cfg.aciklama)}</Text> : null}
                    </View>
                    {maliyet != null ? (
                      <TouchableOpacity
                        style={s.butonKucuk}
                        onPress={() => anaYukseltAl(anahtar as UpgradeType)}
                        activeOpacity={0.8}
                      >
                        <Text style={s.butonKucukYazi}>{para(Number(maliyet))}</Text>
                      </TouchableOpacity>
                    ) : (
                      <View style={[s.butonKucuk, s.butonKucukPasif]}>
                        <Text style={s.butonKucukYazi}>MAX</Text>
                      </View>
                    )}
                  </View>
                );
              })}
            </View>
          ) : null}

          {/* === SPONSORLAR (tesis genelinde) — HER ZAMAN GÖRÜNÜR === */}
          <View style={s.kutu}>
            <View style={s.sponsorBaslikSatir}>
              <Text style={s.kutuBaslik}>💼 Sponsorlar ({imzaliSponsorlar.length}/{MAX_SPONSOR})</Text>
              {imzaliSponsorlar.length > 0 ? (
                <Text style={s.sponsorGelir}>+{para(toplamSponsorGeliri(imzaliSponsorlar))}/gün</Text>
              ) : null}
            </View>
            <Text style={s.muted}>
              Tesis genelinde anlaşma — imzaladığın sponsorlar sahadaki reklam bantlarına yazılır,
              günlük geliri gün sonunda otomatik yatar. Aynı anda en fazla {MAX_SPONSOR} sponsor.
            </Text>
            <View style={s.sponsorIzgara}>
              {SPONSORLAR.map(sp => {
                const imzali = imzaliSponsorlar.includes(sp.id);
                const uygun = sponsorUygunMu(sp, anaSahaAsama, anaSahaUp);
                const nakitYetiyor = state.nakit >= sp.onOdeme;
                return (
                  <View
                    key={sp.id}
                    style={[
                      s.sponsorKart,
                      imzali && { borderColor: sp.renk },
                      !imzali && !uygun && s.sponsorKartKilitli,
                    ]}
                  >
                    <View style={s.sponsorSatir}>
                      <View style={[s.sponsorNokta, { backgroundColor: sp.renk }]} />
                      <Text style={s.sponsorAd} numberOfLines={1}>{sp.ad}</Text>
                    </View>
                    <Text style={s.sponsorAciklama} numberOfLines={2}>{sp.aciklama}</Text>
                    <Text style={s.sponsorPara}>
                      Peşin {para(sp.onOdeme)} · Günlük +{para(sp.gunlukGelir)}
                    </Text>
                    {sponsorKosulSatirlari(sp, anaSahaAsama, anaSahaUp).map((satir, i) => (
                      <Text key={i} style={s.sponsorKosul}>{satir}</Text>
                    ))}
                    {imzali ? (
                      <TouchableOpacity
                        style={[s.sponsorButon, s.sponsorButonKirmizi]}
                        onPress={() => sponsorFesih(sp.id)}
                        activeOpacity={0.8}
                      >
                        <Text style={s.sponsorButonYazi}>Fesih Et — gelir durur</Text>
                      </TouchableOpacity>
                    ) : !uygun ? (
                      <View style={[s.sponsorButon, s.sponsorButonPasif]}>
                        <Text style={s.sponsorButonYazi}>Şartlar sağlanmadı</Text>
                      </View>
                    ) : !nakitYetiyor ? (
                      <View style={[s.sponsorButon, s.sponsorButonPasif]}>
                        <Text style={s.sponsorButonYazi}>Nakit yetersiz (peşin {para(sp.onOdeme)})</Text>
                      </View>
                    ) : (
                      <TouchableOpacity
                        style={s.sponsorButon}
                        onPress={() => sponsorImzala(sp.id)}
                        activeOpacity={0.8}
                      >
                        <Text style={s.sponsorButonYazi}>İmzala — Peşin {para(sp.onOdeme)}</Text>
                      </TouchableOpacity>
                    )}
                  </View>
                );
              })}
            </View>
          </View>
        </View>

        {/* === YATIRIM: YENİ SAHA KATALOĞU === */}
        <Text style={s.bolumBaslik}>Yatırım: Yeni Saha Kataloğu</Text>
        {katalog.length === 0 ? (
          <Text style={s.muted}>Tüm sahalar açıldı — işletme tam kapasitede! 🏆</Text>
        ) : null}
        {katalog.map(d => {
          const tip = SAHA_TIP_BILGI[d.sahaTipi] ?? SAHA_TIP_BILGI.yildiz;
          const nakitYetiyor = state.nakit >= d.arsaBedeli;
          return (
            <View key={d.id} style={[s.katalogKart, { borderColor: `${tip.renk}55` }]}>
              <View style={s.katalogBaslikSatir}>
                <Text style={s.katalogAd}>{tip.emoji} {d.ad}</Text>
                <Text style={s.katalogFiyat}>{para(d.arsaBedeli)}</Text>
              </View>

              {/* Katalogda da 2D önizleme: "bunu alırsan böyle görünecek" */}
              <Saha2D mini asama={0} kalite={d.baslangicKalitesi} sahaTipi={d.sahaTipi} canli={false} />

              <Text style={s.muted}>{d.aciklama}</Text>
              <Text style={s.katalogDetay}>
                ⭐ Başlangıç kalitesi {d.baslangicKalitesi}/10 · Kapasite {d.kapasite} kişi · Günlük gider {para(d.gunlukGider)}
              </Text>
              <Text style={s.katalogDetay}>
                Yatırım: arsa {para(d.arsaBedeli)} + inşaat {para(d.insaatBedeli)} ={' '}
                <Text style={s.sari}>{para(d.arsaBedeli + d.insaatBedeli)}</Text>
              </Text>
              <TouchableOpacity
                style={nakitYetiyor ? s.butonYesil : s.butonPasif}
                onPress={() => arsaAl(d.id)}
                activeOpacity={0.8}
              >
                <Text style={s.butonBeyaz}>
                  {nakitYetiyor
                    ? `Arsayı Al — ${para(d.arsaBedeli)}`
                    : `Nakit yetersiz (${para(d.arsaBedeli)})`}
                </Text>
              </TouchableOpacity>
            </View>
          );
        })}

      </ScrollView>

      {/* === GÜN SONU ÖZETİ (MODAL) === */}
      <Modal visible={!!gunSonu} transparent animationType="fade" onRequestClose={() => setGunSonu(null)}>
        <View style={s.modalKaplama}>
          <View style={s.modalKutu}>
            {gunSonu?.sonuc?.isGameOver ? (
              <Text style={s.modalIflos}>💀 İFLAS ETTİN!</Text>
            ) : (
              <Text style={s.modalBaslik}>🌙 Gün {gunSonu?.sonuc?.gun ?? '?'} Bitti!</Text>
            )}

            <View style={s.modalSatir}>
              <Text style={s.modalEtiket}>💵 Toplam Gelir</Text>
              <Text style={[s.modalDeger, { color: RENK.basari }]}>+{para(gunSonu?.sonuc?.gelir ?? 0)}</Text>
            </View>
            <View style={s.modalSatir}>
              <Text style={s.modalEtiket}>🏟️ Saha Geliri</Text>
              <Text style={s.modalSoluk}>{para(gunSonu?.sonuc?.sahaGelir ?? 0)}</Text>
            </View>
            <View style={s.modalSatir}>
              <Text style={s.modalEtiket}>🧾 Toplam Gider (kira, elektrik, bakım)</Text>
              <Text style={[s.modalDeger, { color: RENK.hata }]}>-{para(gunSonu?.sonuc?.gider ?? 0)}</Text>
            </View>
            <View style={s.modalSatir}>
              <Text style={s.modalEtiket}>👥 Personel Maaşı (giderin içinde)</Text>
              <Text style={s.modalSoluk}>{para(gunSonu?.sonuc?.personelMaas ?? 0)}</Text>
            </View>
            <View style={s.modalSatir}>
              <Text style={s.modalEtiket}>💼 Sponsor Geliri (kasaya yattı)</Text>
              <Text style={[s.modalDeger, { color: RENK.basari }]}>+{para(gunSonu?.sponsorGeliri ?? 0)}</Text>
            </View>

            {(gunSonu?.sonuc?.gorevOdul ?? 0) > 0 ? (
              <View style={s.modalSatir}>
                <Text style={s.modalEtiket}>🎁 Görev Ödülü</Text>
                <Text style={[s.modalDeger, { color: RENK.basari }]}>+{para(gunSonu.sonuc.gorevOdul)}</Text>
              </View>
            ) : null}
            {(gunSonu?.sonuc?.jetonOdul ?? 0) > 0 ? (
              <View style={s.modalSatir}>
                <Text style={s.modalEtiket}>🎟️ Jeton Ödülü</Text>
                <Text style={[s.modalDeger, { color: RENK.basari }]}>+{gunSonu.sonuc.jetonOdul} jeton</Text>
              </View>
            ) : null}

            <View style={s.modalAyirici} />

            <View style={s.modalSatir}>
              <Text style={s.modalEtiket}>📊 Net Kâr / Zarar</Text>
              <Text style={[s.modalDeger, { color: (gunSonu?.sonuc?.net ?? 0) >= 0 ? RENK.basari : RENK.hata }]}>
                {para(gunSonu?.sonuc?.net ?? 0)}
              </Text>
            </View>
            <View style={s.modalSatir}>
              <Text style={s.modalEtiket}>💰 Kasadaki Para</Text>
              <Text style={s.modalKasa}>
                {para((gunSonu?.sonuc?.nakit ?? 0) + (gunSonu?.sponsorGeliri ?? 0))}
              </Text>
            </View>
            <View style={s.modalSatir}>
              <Text style={s.modalEtiket}>😊 Memnuniyet</Text>
              <Text style={s.modalSoluk}>%{gunSonu?.sonuc?.yeniMemnuniyet ?? '—'}</Text>
            </View>
            {(gunSonu?.sonuc?.sadikKazanci ?? 0) > 0 ? (
              <View style={s.modalSatir}>
                <Text style={s.modalEtiket}>❤️ Sadık Müşteri</Text>
                <Text style={[s.modalDeger, { color: RENK.basari }]}>+{gunSonu.sonuc.sadikKazanci}</Text>
              </View>
            ) : null}

            {gunSonu?.sonuc?.isGameOver ? (
              <Text style={s.modalIflosNot}>
                Nakit eksiye düştü ve iflas limiti aşıldı. Ana ekrandan oyunu sıfırlayıp yeniden başlayabilirsin.
              </Text>
            ) : null}

            <TouchableOpacity style={s.modalKapat} onPress={() => setGunSonu(null)} activeOpacity={0.85}>
              <Text style={s.modalKapatYazi}>☀️ Yeni Güne Başla</Text>
            </TouchableOpacity>
          </View>
        </View>
      </Modal>
    </View>
  );
}

function para(n: number): string {
  return `${Math.round(n).toLocaleString('tr-TR')}₺`;
}

function asamaEtiketi(asama: number): string {
  if (asama <= 0) return '🌱 Arsa Yok';
  if (asama === 1) return '🏠 Arsa Alındı';
  if (asama < 5) return '🚧 İnşaat';
  return '✅ Hizmette';
}

const SONUC_ETIKET: Record<TurnuvaResult['sonuc'], string> = {
  birincilik: '🥇 Birincilik',
  ikincilik: '🥈 İkincilik',
  ucunculuk: '🥉 Üçüncülük',
  katilim: 'Katılım',
};

// Personel ücret kuralı — GameContext.hireStaff ile BİREBİR aynı:
// ilk alım = L1'in günlük maaşı, terfi = hedef kademenin günlük maaşı (peşin).
// Günlük maaş ayrıca her gün sonunda düşer.
function staffUcreti(cfg: { levels?: any[] }, currentLevel: number): number {
  if (currentLevel <= 0) return (cfg?.levels?.[0]?.maas) ?? 0;
  return (cfg?.levels?.[currentLevel]?.maas) ?? 0;
}

// Günlük personel maaşı: STAFF_CONFIG şemasından (levels[].maas)
function toplamPersonelMaas(staff: Record<string, StaffMember>): number {
  let toplam = 0;
  for (const s of Object.values(staff ?? {})) {
    if (!s?.hired || s.level < 1) continue;
    const cfg = STAFF_CONFIG.find(c => c.id === s.id);
    toplam += cfg?.levels?.[s.level - 1]?.maas ?? 0;
  }
  return toplam;
}

// Personelin oyuna gerçek etkisi — gameLogic'teki seviye tablolarından okunur (birebir aynı değerler):
// temizlikçi → L1: +5 | L2: +8 | L3: +12 memnuniyet
// kasiyer → L1: -2 | L2: -3 | L3: -4 enerji
// antrenör → L1: +5 | L2: +10 | L3: +15 turnuva performansı
// güvenlik → L1: +1 | L2: +2 | L3: +3 ekstra teklif/gün + olay riski -%10/kademe
function personelEtkiYazisi(id: string, level: number): string | null {
  const seviye = Math.max(0, Math.min(3, Math.round(level ?? 0)));
  if (id === 'temizlikci') {
    const bonus = TEMIZLIKCI_MEMNUNIYET_BONUSU[seviye] ?? 0;
    return bonus > 0 ? `gün sonunda +${bonus} memnuniyet` : null;
  }
  if (id === 'kasiyer') {
    const indirim = KASIYER_ENERJI_INDIRIMI[seviye] ?? 0;
    return indirim > 0 ? `rezervasyonlarda -${indirim} enerji` : null;
  }
  if (id === 'antrenor') {
    const bonus = ANTRENOR_TURNUVA_BONUSU[seviye] ?? 0;
    return bonus > 0 ? `turnuvalarda +${bonus} performans` : null;
  }
  if (id === 'guvenlik') {
    const eki = GUVENLIK_TEKLIF_BONUSU[seviye] ?? 0;
    if (eki <= 0) return null;
    return `gün içinde +${eki} ekstra teklif · olay riski -%${10 * seviye}`;
  }
  return null;
}

// NaN güvenli sayı
function sayi(v: any): number {
  return Number.isFinite(Number(v)) ? Number(v) : 0;
}

// Olay seçeneğinin etki özeti: butonun altında tek satır
function olayEtkiOzeti(etki: any): string {
  if (!etki) return '';
  const p: string[] = [];
  if (etki.sans) p.push(`🎲 %${etki.sans.yuzde} şans`);
  if (etki.nakit) p.push(`💰 ${etki.nakit > 0 ? '+' : ''}${para(etki.nakit)}`);
  if (etki.memnuniyet) p.push(`😊 ${etki.memnuniyet > 0 ? '+' : ''}${etki.memnuniyet}`);
  if (etki.prestij) p.push(`⭐ ${etki.prestij > 0 ? '+' : ''}${etki.prestij}`);
  if (etki.enerji) p.push(`⚡ ${etki.enerji > 0 ? '+' : ''}${etki.enerji}`);
  if (etki.jeton) p.push(`🪙 ${etki.jeton > 0 ? '+' : ''}${etki.jeton}`);
  if (etki.gelirCarpani) p.push(`📈 gelir x${etki.gelirCarpani}`);
  return p.join(' · ');
}

              Saha toplamı: -{para(toplamSahaGideri)} gider · +{para(toplamPasif)} pasif gelir
            </Text>
          </View>
        </View>

        {/* === SAHİBDEN SATIN AL === */}
        {state.yerSahibi ? (
          <View style={[is.kart, is.sahibinKart]}>
            <Text style={is.sahibinBaslik}>🏠 Sahamız Artık Bizim!</Text>
            <Text style={is.sahibinAciklama}>
              Günlük {para(DAILY_KIRA)} kira tamamen kalktı — artık sadece {para(DAILY_ELEKTRIK_SU)} elektrik-su
              ödüyoruz. +{ISLETME_YERI.prestijOdul} prestij kazandın, bu rozet kalıcı. 🎉
            </Text>
          </View>
        ) : (
          <View style={is.kart}>
            <Text style={is.bolumBaslik}>🏷️ Sahibinden Satın Al</Text>
            <Text style={is.sahibinAciklama}>{ISLETME_YERI.aciklama}</Text>
            <View style={is.sahibinSatir}>
              <Text style={is.sahibinEtiket}>💵 Fiyat</Text>
              <Text style={is.sahibinDeger}>{para(yerFiyat)}</Text>
            </View>
            <View style={is.sahibinSatir}>
              <Text style={is.sahibinEtiket}>🪙 Ödül</Text>
              <Text style={is.sahibinDeger}>+{ISLETME_YERI.prestijOdul} prestij</Text>
            </View>
            <View style={is.sahibinSatir}>
              <Text style={is.sahibinEtiket}>✂️ Kaldırır</Text>
              <Text style={is.sahibinDeger}>{para(DAILY_KIRA)}/gün kira</Text>
            </View>
            <TouchableOpacity
              style={[is.buton, { backgroundColor: yereNakitYeterli ? '#22C55E' : '#334155' }]}
              onPress={handleSahibinden}
            >
              <Text style={is.butonYazi}>
                {yereNakitYeterli ? `Yeri Al — ${para(yerFiyat)}` : `Nakit yetersiz (${para(yerFiyat)})`}
              </Text>
            </TouchableOpacity>
          </View>
        )}

        {/* === TESİS YÜKSELTMELERİ === */}
        <View style={is.kart}>
          <Text style={is.bolumBaslik}>⬆️ Tesis Yükseltmeleri</Text>
          {upgradeListe.map(([id, cfg]) => {
            const seviye = state.upgrades?.[id] ?? 0;
            const maks = cfg?.maxLevel ?? cfg?.costs?.length ?? 0;
            const cost = cfg?.costs?.[seviye];
            const makslandi = seviye >= maks;
            return (
              <View key={id} style={is.upSatiri}>
                <Text style={is.upEmoji}>{cfg?.emoji ?? '🔧'}</Text>
                <View style={is.upBilgi}>
                  <Text style={is.upAd}>
                    {cfg?.ad ?? cfg?.isim ?? id} <Text style={is.upSeviye}>L{seviye}/{maks}</Text>
                  </Text>
                  <Text style={is.upAciklama} numberOfLines={2}>{cfg?.aciklama ?? cfg?.etki ?? ''}</Text>
                </View>
                <TouchableOpacity
                  disabled={makslandi}
                  style={[is.upButon, makslandi && { backgroundColor: '#334155' }]}
                  onPress={() => setConfirmUpgrade(id)}
                >
                  <Text style={is.upButonYazi}>{makslandi ? 'MAX' : para(cost ?? 0)}</Text>
                </TouchableOpacity>
              </View>
            );
          })}
        </View>

        {/* === PERSONEL === */}
        <View style={is.kart}>
          <Text style={is.bolumBaslik}>👥 Personel</Text>
          <Text style={is.personelOzet}>
            {kadroSayisi > 0
              ? `${kadroSayisi} kişi görevde · günlük maaş toplamı ${para(personelMaas)} — maaşlar her gün sonunda otomatik düşer. Etkiler seviyeyle ölçeklenir.`
              : 'Kadro boş: temizlikçi memnuniyeti yükseltir, kasiyer rezervasyonlarda enerji tasarrufu sağlar. Maaşlar gün sonunda düşer.'}
          </Text>
          {STAFF_CONFIG.map(cfg => {
            const m = state.staff?.[cfg.id] ?? { id: cfg.id, hired: false, level: 0 };
            const seviye = m.level ?? 0;
            const maks = cfg.levels?.length ?? 0;
            const makslandi = seviye >= maks;
            const ucret = staffUcreti(cfg, seviye);
            const sonraki = makslandi ? null : cfg.levels?.[seviye];
            const maas = sonraki?.maas ?? 0;
            const etkiYazisi = personelEtkiYazisi(cfg.id, seviye);
            return (
              <View key={cfg.id} style={is.upSatiri}>
                <Text style={is.upEmoji}>{cfg.emoji ?? '🧑‍💼'}</Text>
                <View style={is.upBilgi}>
                  <Text style={is.upAd}>
                    {cfg.label ?? cfg.id} <Text style={is.upSeviye}>L{seviye}/{maks}</Text>
                  </Text>
                  <Text style={is.upAciklama} numberOfLines={2}>
                    {cfg.aciklama ?? ''}{sonraki ? ` · Sonraki: ${sonraki.label ?? '—'} (${para(maas)}/gün)` : ''}
                  </Text>
                  {seviye > 0 && (
                    <Text style={is.personelEtki}>
                      {etkiYazisi
                        ? `✓ Aktif: ${etkiYazisi}`
                        : `✓ Görevde: ${cfg.levels?.[seviye - 1]?.label ?? 'Aktif'}`}
                    </Text>
                  )}
                </View>
                <TouchableOpacity
                  disabled={makslandi}
                  style={[is.upButon, makslandi && { backgroundColor: '#334155' }]}
                  onPress={() => setConfirmStaff(cfg.id)}
                >
                  <Text style={is.upButonYazi}>
                    {makslandi ? 'MAX' : seviye === 0 ? `Al ${para(ucret)}` : `Terfi ${para(ucret)}`}
                  </Text>
                </TouchableOpacity>
              </View>
            );
          })}
        </View>

        {/* === TURNUVA === */}
        <View style={is.kart}>
          <Text style={is.bolumBaslik}>🏆 Turnuva</Text>
          <Text style={is.sahibinAciklama}>
            Halısaha turnuvasına katıl, ödül ve prestij kazan! Katılım: {para(katilimBedeli)} · Her {aralikGun} günde bir.
          </Text>
          {!turnuvaReady ? (
            <View style={is.turnuvaBekle}>
              <Text style={is.turnuvaBekleYazi}>⏳ Sonraki turnuvaya {kalanGun} gün</Text>
            </View>
          ) : (
            <TouchableOpacity style={[is.buton, { backgroundColor: '#F59E0B' }]} onPress={handleTurnuva}>
              <Text style={is.butonKoyu}>Turnuvaya Katıl — {para(katilimBedeli)}</Text>
            </TouchableOpacity>
          )}
        </View>

        {/* === GÜNÜN OLAYI — soru ve seçenekler doğrudan burada === */}
        <View style={is.kart}>
          <Text style={is.bolumBaslik}>🎲 Olay Durumu</Text>
          {bugununOlayi ? (
            <>
              <Text style={is.olayBaslik}>{bugununOlayi.emoji} {bugununOlayi.baslik}</Text>
              <Text style={is.olayYazi}>{bugununOlayi.aciklama}</Text>
              {state.olayCozuldu ? (
                olaySonucMesaji
                  ? <Text style={is.olaySonuc}>📌 {olaySonucMesaji}</Text>
                  : <Text style={is.olaySonuc}>📌 Bugünün olayı çözüldü.</Text>
              ) : (
                <>
                  <Text style={is.olaySoru}>Ne yapıyorsun?</Text>
                  {(bugununOlayi.secenekler ?? []).map((sec, i) => (
                    <TouchableOpacity
                      key={`${bugununOlayi.id}_${i}`}
                      style={is.olaySecenek}
                      onPress={() => handleOlaySec(i)}
                    >
                      <Text style={is.olaySecenekLabel}>{sec.label}</Text>
                      <Text style={is.olaySecenekAciklama}>{sec.aciklama}</Text>
                      {olayEtkiOzeti(sec.etki).length > 0 && (
                        <Text style={is.olaySecenekEtki}>{olayEtkiOzeti(sec.etki)}</Text>
                      )}
                    </TouchableOpacity>
                  ))}
                </>
              )}
            </>
          ) : (
            <Text style={is.olayYazi}>☀️ Bugün sakin bir gün — olay yok. (Olay her gün çıkmaz; güvenlik kademesi çıkma şansını düşürür.)</Text>
          )}
          {korumaListesi.length > 0 && (
            <Text style={is.olayKoruma}>{korumaListesi.join(' · ')}</Text>
          )}
          {olayGecmisListesi.length > 0 && (
            <>
              <Text style={is.olayGecmisBaslik}>Son olaylar:</Text>
              {olayGecmisListesi.map((o, i) => (
                <Text key={`${o.id}_${i}`} style={is.olayGecmisSatir}>{o.emoji} {o.baslik}</Text>
              ))}
            </>
          )}
        </View>

        {/* === LİG === */}
        <View style={is.kart}>
          <Text style={is.bolumBaslik}>⚽ Beşli Patron Ligi</Text>
          {!lig ? (
            <>
              <Text style={is.sahibinAciklama}>
                8 takımlı, tek devreli lig (7 hafta). Günde en fazla 1 hafta maçı ({LIG_CONFIG.macEnerjisi} enerji).
                Galibiyet {para(LIG_CONFIG.oduller.galibiyet)} + prestij. Sezonu ilk 3'te bitir, sezon ödülü kap.
                Takım gücün: saha kalitesi, antrenör, memnuniyet ve prestijden beslenir.
              </Text>
              <TouchableOpacity
                style={[is.buton, { backgroundColor: yildizAcik ? '#F59E0B' : '#334155' }]}
                onPress={handleLigKatil}
              >
                <Text style={is.butonYazi}>{yildizAcik ? 'Lige Katıl — Ücretsiz' : 'Yıldız saha hizmette olmalı'}</Text>
              </TouchableOpacity>
            </>
          ) : (
            <>
              <View style={is.ligBaslikSatir}>
                <Text style={is.ligSezonYazi}>🏆 {lig.sezon}. Sezon · Hafta {lig.hafta}/7</Text>
                {lig.sampiyonluk > 0 && (
                  <Text style={is.ligSampiyonluk}>⭐ {lig.sampiyonluk}x şampiyon</Text>
                )}
              </View>

              {lig.sonMac && (
                <Text style={is.ligSonMac}>
                  Son maç: {lig.sonMac.senGol}-{lig.sonMac.rakipGol} {lig.sonMac.rakipAd}{' '}
                  {lig.sonMac.sonuc === 'galibiyet' ? '✅' : lig.sonMac.sonuc === 'beraberlik' ? '➖' : '❌'}
                </Text>
              )}

              {/* Puan durumu */}
              <View style={is.ligTabloBaslik}>
                <Text style={[is.ligTh, { flex: 0.5 }]}>#</Text>
                <Text style={[is.ligThAd, { flex: 2.3 }]}>TAKIM</Text>
                <Text style={[is.ligTh, { flex: 0.7 }]}>O</Text>
                <Text style={[is.ligTh, { flex: 0.8 }]}>AV</Text>
                <Text style={[is.ligTh, { flex: 0.7 }]}>P</Text>
              </View>
              {ligTablo.map((t, i) => (
                <View key={t.id} style={[is.ligSatir, t.id === 'ben' && is.ligSatirBen]}>
                  <Text style={[is.ligTd, { flex: 0.5 }]}>{i + 1}</Text>
                  <Text style={[is.ligTdAd, { flex: 2.3 }]} numberOfLines={1}>{t.ad}</Text>
                  <Text style={[is.ligTd, { flex: 0.7 }]}>{t.oynanan}</Text>
                  <Text style={[is.ligTd, { flex: 0.8 }]}>
                    {t.attigi - t.yedigi > 0 ? '+' : ''}{t.attigi - t.yedigi}
                  </Text>
                  <Text style={[is.ligTdPuan, { flex: 0.7 }]}>{t.puan}</Text>
                </View>
              ))}

              {rakipLigTakim && (
                <Text style={is.ligSirada}>
                  Sırada: {rakipLigTakim.ad} (güç {rakipLigTakim.guc}) · Senin güç: {benimLigTakim?.guc ?? '?'}
                </Text>
              )}
              <TouchableOpacity
                style={[is.buton, { backgroundColor: ligMacAcik ? '#F59E0B' : '#334155' }]}
                onPress={handleLigMac}
              >
                <Text style={is.butonYazi}>
                  {ligMacAcik
                    ? `${lig.hafta}. Hafta Maçı Oyna — ${LIG_CONFIG.macEnerjisi} enerji`
                    : 'Bugünün maçı oynandı ✅'}
                </Text>
              </TouchableOpacity>
              <TouchableOpacity style={is.ligCikBtn} onPress={handleLigBirak}>
                <Text style={is.ligCikYazi}>Ligden ayrıl</Text>
              </TouchableOpacity>
            </>
          )}
        </View>
      </ScrollView>

      {/* === YÜKSELTME ONAY MODALI === */}
      <Modal visible={confirmUpgrade !== null} transparent animationType="fade" onRequestClose={() => setConfirmUpgrade(null)}>
        <View style={is.modalOverlay}>
          <View style={is.modalKart}>
            <Text style={is.modalBaslik}>{onayCfg?.emoji ?? '🔧'} {onayCfg?.ad ?? onayCfg?.isim ?? 'Yükseltme'}</Text>
            <Text style={is.modalYazi}>
              L{onaySeviye} → L{onaySeviye + 1} yükseltmesi için {para(onayCost)} ödenecek. Onaylıyor musun?
            </Text>
            <View style={is.modalButonlar}>
              <TouchableOpacity style={is.btnIptal} onPress={() => setConfirmUpgrade(null)}>
                <Text style={is.btnIptalYazi}>Vazgeç</Text>
              </TouchableOpacity>
              <TouchableOpacity style={is.btnOnay} onPress={handleUpgradeOnay}>
                <Text style={is.btnOnayYazi}>Onayla</Text>
              </TouchableOpacity>
            </View>
          </View>
        </View>
      </Modal>

      {/* === PERSONEL ONAY MODALI === */}
      <Modal visible={confirmStaff !== null} transparent animationType="fade" onRequestClose={() => setConfirmStaff(null)}>
        <View style={is.modalOverlay}>
          <View style={is.modalKart}>
            <Text style={is.modalBaslik}>{onayStaffCfg?.emoji ?? '🧑‍💼'} {onayStaffCfg?.label ?? 'Personel'}</Text>
            <Text style={is.modalYazi}>
              {onayStaffLevel === 0 ? 'İşe alım' : 'Terfi'} ücreti {para(onayStaffUcret)}. Onaylıyor musun?
            </Text>
            <View style={is.modalButonlar}>
              <TouchableOpacity style={is.btnIptal} onPress={() => setConfirmStaff(null)}>
                <Text style={is.btnIptalYazi}>Vazgeç</Text>
              </TouchableOpacity>
              <TouchableOpacity style={is.btnOnay} onPress={handleStaffOnay}>
                <Text style={is.btnOnayYazi}>Onayla</Text>
              </TouchableOpacity>
            </View>
          </View>
        </View>
      </Modal>

      {/* === TURNUVA SONUÇ MODALI === */}
      <Modal visible={turnuvaResult !== null} transparent animationType="fade" onRequestClose={() => setTurnuvaResult(null)}>
        <View style={is.modalOverlay}>
          <View style={is.modalKart}>
            <Text style={is.modalBaslik}>🏆 {turnuvaResult?.isim ?? 'Turnuva'}</Text>
            <Text style={is.modalSonuc}>{turnuvaResult ? SONUC_ETIKET[turnuvaResult.sonuc] : ''}</Text>
            <Text style={is.modalYazi}>
              💰 Kazanç: {para(turnuvaResult?.kazanc ?? 0)}{'\n'}⭐ Prestij: +{turnuvaResult?.prestijKazanc ?? 0}
            </Text>
            <View style={is.modalButonlar}>
              <TouchableOpacity style={is.btnOnay} onPress={() => setTurnuvaResult(null)}>
                <Text style={is.btnOnayYazi}>Harika!</Text>
              </TouchableOpacity>
            </View>
          </View>
        </View>
      </Modal>

      {/* === LİG MAÇ SONUCU MODALI === */}
      <Modal visible={ligSonuc !== null} transparent animationType="fade" onRequestClose={() => setLigSonuc(null)}>
        <View style={is.modalOverlay}>
          <View style={is.modalKart}>
            <Text style={is.modalBaslik}>⚽ {ligSonuc?.hafta}. Hafta</Text>
            <Text style={is.ligModalSkor}>
              Beşli Patron SK {ligSonuc?.senGol} - {ligSonuc?.rakipGol} {ligSonuc?.rakipAd}
            </Text>
            <Text style={is.modalSonuc}>
              {ligSonuc?.sonuc === 'galibiyet'
                ? '✅ Galibiyet!'
                : ligSonuc?.sonuc === 'beraberlik'
                  ? '➖ Beraberlik'
                  : '❌ Yenilgi'}
            </Text>
            {ligSonuc && ligSonuc.sezonBitti && (
              <Text style={is.ligModalSezon}>
                🏁 Sezon bitti! Final sırası: {ligSonuc.siralama}.{' '}
                {ligSonuc.siralama === 1 ? 'ŞAMPİYON! 🎉' : ''}
              </Text>
            )}
            <Text style={is.modalYazi}>
              💰 Maç kazancı: +{para(ligSonuc?.kazanc ?? 0)}{'\n'}
              {(ligSonuc?.sezonOdulu ?? 0) > 0
                ? `🏆 Sezon ödülü: +${para(ligSonuc?.sezonOdulu ?? 0)}\n`
                : ''}
              ⭐ Prestij: +{(ligSonuc?.prestij ?? 0) + (ligSonuc?.sezonPrestij ?? 0)}
            </Text>
            <View style={is.modalButonlar}>
              <TouchableOpacity style={is.btnOnay} onPress={() => setLigSonuc(null)}>
                <Text style={is.btnOnayYazi}>Tamam</Text>
              </TouchableOpacity>
            </View>
          </View>
        </View>
      </Modal>

      {/* === OLAY SONUCU MODALI === */}
      <Modal visible={olaySonucAcik} transparent animationType="fade" onRequestClose={() => setOlaySonucAcik(false)}>
        <View style={is.modalOverlay}>
          <View style={is.modalKart}>
            <Text style={is.modalBaslik}>📌 Olay Sonucu</Text>
            <Text style={is.modalSonuc}>{olaySonucMesaji ?? 'Karar uygulandı.'}</Text>
            {olayOzet.length > 0 && <Text style={is.modalYazi}>{olayOzet}</Text>}
            <View style={is.modalButonlar}>
              <TouchableOpacity style={is.btnOnay} onPress={() => setOlaySonucAcik(false)}>
                <Text style={is.btnOnayYazi}>Tamam</Text>
              </TouchableOpacity>
            </View>
          </View>
        </View>
      </Modal>

      {/* === BİLDİRİM === */}
      {mesaj != null && (
        <View style={[is.toast, { backgroundColor: hata ? '#7F1D1D' : '#14532D' }]}>
          <Text style={is.toastYazi}>{mesaj}</Text>
          <TouchableOpacity onPress={() => setMesaj(null)}>
            <Text style={is.toastYazi}>✕</Text>
          </TouchableOpacity>
        </View>
      )}
    </View>
  );
}

import { LIFE_ITEMS, KIZ_ARKADAS_CONFIG, type LifeItemConfig } from '../../constants/gameConfig';

const KATEGORI_LABELS: Record<string, { emoji: string; label: string }> = {
  ulasim: { emoji: '🚗', label: 'Ulaşım' },
  konut: { emoji: '🏠', label: 'Konut' },
  luks: { emoji: '✨', label: 'Lüks' },
};

export default function YasamScreen() {
  const { state, buyLifeItem, upgradeKizArkadas } = useGame();
  const [message, setMessage] = useState('');
  const [msgType, setMsgType] = useState<'success' | 'error'>('success');
  const [confirmItem, setConfirmItem] = useState<{ type: 'life'; id: string } | { type: 'kiz' } | null>(null);
  const font = Platform.select({ ios: 'SpaceMono', android: 'SpaceMono', web: 'SpaceMono, monospace' });

  const prestij = state?.prestij ?? 0;

  const kizLevels = (KIZ_ARKADAS_CONFIG?.levels ?? []) as { emoji: string; label: string; cost: number; prestij: number }[];
  const kizRequiredPrestij = KIZ_ARKADAS_CONFIG?.requiredPrestij ?? 0;
  const lifeItems = LIFE_ITEMS ?? [];

  const showMsg = (msg: string, type: 'success' | 'error' = 'success') => {
    setMessage(msg);
    setMsgType(type);
    setTimeout(() => setMessage(''), 2500);
  };

  const handleConfirm = () => {
    if (!confirmItem) return;
    if (confirmItem.type === 'life') {
      const err = buyLifeItem(confirmItem.id);
      if (err) showMsg(err, 'error');
      else showMsg('Satın alındı! 🎉');
    } else {
      const err = upgradeKizArkadas();
      if (err) showMsg(err, 'error');
      else showMsg('Ela ilişkiyi ilerletti! ❤️');
    }
    setConfirmItem(null);
  };

  const kategoriler = ['ulasim', 'konut', 'luks'] as const;

  const kaSeviye = state?.kizArkdasSeviye ?? 0;
  const kaMaxed = kaSeviye >= kizLevels.length;
  const kaNextLevel = kaMaxed ? null : kizLevels[kaSeviye];
  const kaCanBuy = !kaMaxed && !!kaNextLevel && (state?.nakit ?? 0) >= (kaNextLevel?.cost ?? 0) && prestij >= kizRequiredPrestij;
  const kaPrestijLocked = prestij < kizRequiredPrestij;

  return (
    <ScrollView style={styles.container} contentContainerStyle={styles.content}>
      {message ? (
        <View style={[styles.messageBox, msgType === 'error' && styles.messageError]}>
          <Text style={[styles.messageText, { fontFamily: font }, msgType === 'error' && styles.messageTextError]}>{message}</Text>
        </View>
      ) : null}

      {/* Prestij Bar */}
      <View style={styles.prestijBar}>
        <Text style={[styles.prestijLabel, { fontFamily: font }]}>⭐ Prestij Puanı</Text>
        <Text style={[styles.prestijValue, { fontFamily: font }]}>{prestij}</Text>
      </View>

      {/* Kategori sections */}
      {kategoriler.map(kat => {
        const katInfo = KATEGORI_LABELS[kat];
        const items = lifeItems.filter(i => i?.kategori === kat);
        return (
          <View key={kat} style={styles.section}>
            <Text style={[styles.sectionTitle, { fontFamily: font }]}>{katInfo?.emoji} {katInfo?.label}</Text>
            {items.map(item => {
              const owned = state?.lifeItems?.[item.id];
              const currentLevel = owned?.level ?? 0;
              const maxed = currentLevel >= (item.levels?.length ?? 0);
              const nextLevel = maxed ? null : item.levels?.[currentLevel];
              const canAfford = nextLevel ? (state?.nakit ?? 0) >= (nextLevel.cost ?? 0) : false;
              const prestijLocked = prestij < (item.requiredPrestij ?? 0);
              const canBuy = !maxed && canAfford && !prestijLocked;

              return (
                <View key={item.id} style={[styles.itemCard, prestijLocked && styles.itemLocked]}>
                  <View style={styles.itemHeader}>
                    <Text style={styles.itemEmoji}>{item.emoji}</Text>
                    <View style={styles.itemInfo}>
                      <Text style={[styles.itemName, { fontFamily: font }]}>{item.label}</Text>
                      <Text style={[styles.itemDesc, { fontFamily: font }]}>{item.aciklama}</Text>
                      {currentLevel > 0 && (
                        <Text style={[styles.itemCurrentLevel, { fontFamily: font }]}>
                          ✅ {item.levels?.[currentLevel - 1]?.label ?? ''}
                        </Text>
                      )}
                    </View>
                  </View>
                  {prestijLocked ? (
                    <View style={styles.lockedRow}>
                      <Text style={[styles.lockedText, { fontFamily: font }]}>🔒 {item.requiredPrestij ?? 0} Prestij gerekli</Text>
                    </View>
                  ) : maxed ? (
                    <View style={styles.maxedRow}>
                      <Text style={[styles.maxedText, { fontFamily: font }]}>✅ Maksimum</Text>
                    </View>
                  ) : nextLevel ? (
                    <View style={styles.buyRow}>
                      <View>
                        <Text style={[styles.nextLevelLabel, { fontFamily: font }]}>{nextLevel.label}</Text>

                      </View>
                      <Pressable

                        onPress={() => setConfirmItem({ type: 'life', id: item.id })}
                        disabled={!canBuy}

                        accessibilityRole="button"
                      >
                        <Text style={[styles.buyBtnText, { fontFamily: font }]}>Satın Al</Text>
                      </Pressable>
                    </View>
                  ) : null}
                </View>
              );
            })}
          </View>
        );
      })}

      {/* Ela */}
      <View style={styles.section}>
        <Text style={[styles.sectionTitle, { fontFamily: font }]}>❤️ İlişki</Text>
        <View style={[styles.itemCard, kaPrestijLocked && styles.itemLocked]}>
          <View style={styles.itemHeader}>
            <Text style={styles.itemEmoji}>👩</Text>
            <View style={styles.itemInfo}>
              <Text style={[styles.itemName, { fontFamily: font }]}>Ela 💕</Text>
              <Text style={[styles.itemDesc, { fontFamily: font }]}>Hayatına Ela'yı kat</Text>
              {kaSeviye > 0 && (
                <Text style={[styles.itemCurrentLevel, { fontFamily: font }]}>
                  {kizLevels[kaSeviye - 1]?.emoji ?? ''} {kizLevels[kaSeviye - 1]?.label ?? ''}
                </Text>

            </View>
          ) : kaMaxed ? (

import React, { useState } from 'react';
import {
  View, Text, ScrollView, TouchableOpacity, StyleSheet,
} from 'react-native';
import { useGame } from '../../context/GameContext';
import { MAX_ENERJI } from '../../constants/gameConfig';
import { SAHA_TIP_BILGI } from '../../constants/pitchConfig';
import { rezervasyonEnerjiMaliyeti } from '../../services/gameLogic';
import type { Reservation, PitchState } from '../../types/game';

function para(n: number): string {
  return `${Math.round(n).toLocaleString('tr-TR')}₺`;
}

// Müşteri tipi → gösterim (sadece ekran süsü, config'e bağımlı değil)
const MUSTERI_BILGI: Record<string, { emoji: string; ad: string }> = {
  mahalle: { emoji: '🏘️', ad: 'Mahalle Takımı' },
  ogrenci: { emoji: '🎓', ad: 'Öğrenci Grubu' },
  ofis: { emoji: '💼', ad: 'Ofis Ekibi' },
  amator: { emoji: '⚽', ad: 'Amatör Lig' },
  dogumGunu: { emoji: '🎂', ad: 'Doğum Günü' },
  yas50plus: { emoji: '👴', ad: '50+ Kafalar' },
  kurumsal: { emoji: '🏢', ad: 'Kurumsal Takım' },
};

export default function RezervasyonScreen() {
  const { state, acceptReservation, rejectReservation } = useGame();
  const [mesaj, setMesaj] = useState<string | null>(null);
  const [hata, setHata] = useState(false);
  const [filtre, setFiltre] = useState<string>('hepsi'); // 'hepsi' | sahaId

  if (!state) {
    return (
      <View style={st.ekran}>
        <Text style={st.sari}>Yükleniyor…</Text>
      </View>
    );
  }

  // === SAHALAR ===
  const sahaListesi = Object.values(state.pitches ?? {}).filter(Boolean) as PitchState[];
  const acikSahalar = sahaListesi.filter(p => (p.asama ?? 0) >= 5);

  // === TEKLİFLER ===
  const tumTeklifler = (state.bugunRezervasyonlar ?? []).filter(r => r?.durum === 'bekliyor');
  const gosterilen = filtre === 'hepsi'
    ? tumTeklifler
    : tumTeklifler.filter(r => (r.sahaId ?? 'yildiz') === filtre);
  const kabulEdilenler = state.kabulEdilenler ?? [];
  const bugunGelir = state.bugunGelir ?? 0;

  const bildir = (sonuc: string | null, basari?: string) => {
    if (sonuc) { setHata(true); setMesaj(sonuc); }
    else if (basari) { setHata(false); setMesaj(basari); }
  };

  // Tek teklifin sahasının rozet bilgisi (eski kayıtlarda sahaId yoksa Yıldız)
  const sahaBilgi = (sahaId?: string) => {
    const id = sahaId ?? 'yildiz';
    const p = state.pitches?.[id];
    const tip = SAHA_TIP_BILGI[p?.sahaTipi ?? 'yildiz'] ?? SAHA_TIP_BILGI.yildiz;
    return { id, ad: p?.ad ?? 'Yıldız Saha', emoji: tip.emoji, renk: tip.renk };
  };

  // Kartta gösterilen enerji maliyeti — Context'in kabul anında yaktığı formülle aynı:
  // grup büyüklüğü + kasiyer indirimi + sahanın kendi turnike indirim (Yıldız'ın yükseltmesi state.upgrades'te)
  const enerjiMaliyeti = (rez: Reservation): number => {
    const id = rez.sahaId ?? 'yildiz';
    const p = state.pitches?.[id];
    const up: Record<string, number> = id === 'yildiz'
      ? ((state.upgrades ?? {}) as Record<string, number>)
      : ((p?.upgrades ?? {}) as Record<string, number>);
    return rezervasyonEnerjiMaliyeti(
      rez.kisiSayisi ?? 0,
      state.staff?.kasiyer?.level ?? 0,
      up?.turnike ?? 0,
    );
  };

  const kabulEt = (rez: Reservation) => {
    const maliyet = enerjiMaliyeti(rez);
    const sonuc = acceptReservation(rez.id);
    if (sonuc) bildir(sonuc);
    else bildir(null, `🤝 ${rez.musteriAdi} kabul edildi — ${sahaBilgi(rez.sahaId).ad}, ${maliyet} enerji.`);
  };

  const reddet = (rez: Reservation) => {
    rejectReservation(rez.id);
    bildir(null, `🚫 ${rez.musteriAdi} reddedildi.`);
  };

  const enerjiYuzde = Math.max(0, Math.min(100, ((state.enerji ?? 0) / MAX_ENERJI) * 100));

  // Hiç açık saha yoksa defter yerine uyarı
  if (acikSahalar.length === 0) {
    return (
      <View style={st.ekran}>
        <ScrollView contentContainerStyle={st.icerik} showsVerticalScrollIndicator={false}>
          <Text style={st.baslik}>📞 Rezervasyon Defteri</Text>
          <View style={st.kart}>
            <Text style={st.bolumBaslik}>🚧 Açık sahan yok</Text>
            <Text style={st.aciklama}>
              Teklif alabilmen için en az bir sahanın inşaatı bitmiş olmalı.
              Saha sekmesinden inşaatı ilerlet — açılan saha için burada teklifler akmaya başlar.
            </Text>
          </View>
        </ScrollView>
        {mesaj != null && (
          <View style={[st.toast, { backgroundColor: hata ? '#7F1D1D' : '#14532D' }]}>
            <Text style={st.toastYazi}>{mesaj}</Text>
            <TouchableOpacity onPress={() => setMesaj(null)}>
              <Text style={st.toastYazi}>✕</Text>
            </TouchableOpacity>
          </View>
        )}
      </View>
    );
  }

  return (
    <View style={st.ekran}>
      <ScrollView contentContainerStyle={st.icerik} showsVerticalScrollIndicator={false}>
        <Text style={st.baslik}>📞 Rezervasyon Defteri</Text>

        {/* === GÜN DURUMU === */}
        <View style={st.kart}>
          <View style={st.durumSatir}>
            <Text style={st.durumDeger}>📅 Gün {state.gun ?? 1}</Text>
            <Text style={st.durumDeger}>⚡ {state.enerji ?? 0}/{MAX_ENERJI}</Text>
            <Text style={st.durumDeger}>💰 {para(state.nakit ?? 0)}</Text>
            <Text style={st.durumDeger}>📈 {para(bugunGelir)}</Text>
          </View>
          <View style={st.barArka}>
            <View style={[st.barOn, { width: `${enerjiYuzde}%` }]} />
          </View>
          <Text style={st.durumNot}>
            Her kabul ettiğin teklif enerji yakar — grup büyüdükçe maliyet artar;
            kasiyerin ve sahanın turnike sistemi indirim yapar.
          </Text>
        </View>

        {/* === SAHA FİLTRESİ === */}
        <ScrollView horizontal showsHorizontalScrollIndicator={false} style={st.seri}>
          <TouchableOpacity
            style={[st.cip, filtre === 'hepsi' && st.cipSecili]}
            onPress={() => { setFiltre('hepsi'); setMesaj(null); }}
          >
            <Text style={st.cipEmoji}>🗂️</Text>
            <Text style={st.cipAd}>Tümü ({tumTeklifler.length})</Text>
          </TouchableOpacity>
          {acikSahalar.map(p => {
            const b = sahaBilgi(p.id);
            const adet = tumTeklifler.filter(r => (r.sahaId ?? 'yildiz') === p.id).length;
            const secili = filtre === p.id;
            return (
              <TouchableOpacity
                key={p.id}
                style={[st.cip, secili && { borderColor: b.renk, backgroundColor: '#1E293B' }]}
                onPress={() => { setFiltre(p.id); setMesaj(null); }}
              >
                <Text style={st.cipEmoji}>{b.emoji}</Text>
                <Text style={st.cipAd}>{p.ad ?? 'Saha'}</Text>
                <Text style={st.cipAdet}>{adet}</Text>
              </TouchableOpacity>
            );
          })}
        </ScrollView>

        {/* === TEKLİF LİSTESİ === */}
        {gosterilen.length === 0 ? (
          <View style={st.kart}>
            <Text style={st.bos}>
              {tumTeklifler.length === 0
                ? '😴 Bugünlük başka teklif yok. Günü bitir, yarın yeni müşteriler gelir.'
                : '📭 Bu saha için bekleyen teklif yok — başka sahayı da kontrol et.'}
            </Text>
          </View>
        ) : (
          gosterilen.map(rez => {
            const sB = sahaBilgi(rez.sahaId);
            const mb = MUSTERI_BILGI[rez.musteriTipi] ?? { emoji: '🙂', ad: 'Müşteri' };
            const maliyet = enerjiMaliyeti(rez);
            const enerjiYetiyor = (state.enerji ?? 0) >= maliyet;
            return (
              <View key={rez.id} style={st.kart}>
                {/* Kart başı: müşteri + saha rozeti */}
                <View style={st.kartBas}>
                  <Text style={st.musteriEmoji}>{mb.emoji}</Text>
                  <View style={{ flex: 1 }}>
                    <Text style={st.musteriAd}>{rez.musteriAdi}</Text>
                    <Text style={st.musteriTur}>{mb.ad}</Text>
                  </View>
                  <View style={[st.sahaRozet, { borderColor: sB.renk }]}>
                    <Text style={st.sahaRozetYazi}>{sB.emoji} {sB.ad}</Text>
                  </View>
                </View>

                {/* Saat + enerji maliyeti */}
                <View style={st.bilgiSatiri}>
                  <Text style={st.bilgiCip}>⏰ {rez.saatAraligi}</Text>
                  <Text style={[st.bilgiCip, !enerjiYetiyor && { color: '#F87171' }]}>⚡ {maliyet} enerji</Text>
                  {rez.sadik ? <Text style={[st.bilgiCip, { color: '#F9A8D4' }]}>❤️ Sadık müşteri</Text> : null}
                </View>

                {/* Fiyat + aksiyonlar */}
                <View style={st.aksiyonSatiri}>
                  <Text style={st.fiyat}>{para(rez.fiyat ?? 0)}</Text>
                  <View style={st.butonGrup}>
                    <TouchableOpacity
                      style={st.btnRed}
                      onPress={() => reddet(rez)}
                      accessibilityRole="button"
                      accessibilityLabel={`${rez.musteriAdi ?? ''} reddet`}
                    >
                      <Text style={st.btnRedYazi}>❌ Reddet</Text>
                    </TouchableOpacity>
                    <TouchableOpacity
                      style={[st.btnYesil, !enerjiYetiyor && { backgroundColor: '#334155' }]}
                      disabled={!enerjiYetiyor}
                      onPress={() => kabulEt(rez)}
                      accessibilityRole="button"
                      accessibilityLabel={`${rez.musteriAdi ?? ''} kabul et`}
                    >
                      <Text style={st.btnYesilYazi}>
                        {enerjiYetiyor ? '✅ Kabul Et' : '⚡ Enerji Az'}
                      </Text>
                    </TouchableOpacity>
                  </View>
                </View>
              </View>
            );
          })
        )}

        {/* === BUGÜN KABUL EDİLENLER === */}
        {kabulEdilenler.length > 0 && (
          <View style={st.kart}>
            <Text style={st.bolumBaslik}>✅ Bugün Kabul Edilenler ({kabulEdilenler.length})</Text>
            {kabulEdilenler.map(r => {
              const sB = sahaBilgi(r.sahaId);
              return (
                <View key={r.id} style={st.kabulSatiri}>
                  <Text style={st.kabulSaha}>{sB.emoji}</Text>
                  <Text style={st.kabulAd} numberOfLines={1}>{r.musteriAdi}</Text>
                  <Text style={st.kabulSaat}>{r.saatAraligi}</Text>
                  <Text style={st.kabulFiyat}>{para(r.fiyat ?? 0)}</Text>
                </View>
              );
            })}
            <Text style={st.kabulToplam}>Toplam {para(bugunGelir)} — gün sonunda kasaya girer.</Text>
          </View>
        )}
      </ScrollView>

      {/* === BİLDİRİM === */}
      {mesaj != null && (
        <View style={[st.toast, { backgroundColor: hata ? '#7F1D1D' : '#14532D' }]}>
          <Text style={st.toastYazi}>{mesaj}</Text>
          <TouchableOpacity onPress={() => setMesaj(null)}>
            <Text style={st.toastYazi}>✕</Text>
          </TouchableOpacity>
        </View>
      )}
    </View>
  );
}
